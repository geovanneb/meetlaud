let langs: [String: NLLanguage] = ["en": .english, "pt": .portuguese, "es": .spanish, "ja": .japanese]

func embedder(_ l: String) -> NLContextualEmbedding {
    let e = NLContextualEmbedding(language: langs[l]!)!
    try! e.load(); return e
}
func vec(_ e: NLContextualEmbedding, _ t: String, _ l: String) -> [Float] {
    let r = try! e.embeddingResult(for: t, language: langs[l]!)
    var s = [Float](repeating: 0, count: e.dimension); var n: Float = 0
    r.enumerateTokenVectors(in: t.startIndex..<t.endIndex) { v, _ in
        for i in 0..<v.count { s[i] += Float(v[i]) }; n += 1; return true }
    var norm: Float = 0; vDSP_svesq(s, 1, &norm, vDSP_Length(s.count)); norm = sqrt(norm)
    return s.map { $0 / max(norm, 1e-6) }
}
func dot(_ a: [Float], _ b: [Float]) -> Float { var r: Float = 0; vDSP_dotpr(a,1,b,1,&r,vDSP_Length(a.count)); return r }

func run(q ql: String, p pl: String) {
    let e = embedder(ql), f = embedder(pl)
    let ps = passages[pl]!.map { vec(f, $0, pl) }
    var r1 = 0, r3 = 0, r5 = 0; var mrr: Float = 0
    for (q, want) in queries[ql]! {
        let v = vec(e, q, ql)
        let order = ps.indices.sorted { dot(v, ps[$0]) > dot(v, ps[$1]) }
        let rank = order.firstIndex(of: want)!
        if rank < 1 {r1 += 1}; if rank < 3 {r3 += 1}; if rank < 5 {r5 += 1}; mrr += 1/Float(rank+1)
    }
    print("query=\(ql) passages=\(pl) same-model=\(e.modelIdentifier == f.modelIdentifier)  r@1 \(r1)/8  r@3 \(r3)/8  r@5 \(r5)/8  mrr \(String(format:"%.2f", mrr/8))")
}
